10-json / 02_parsing_json.html
==================================================

<html>
<head>
    <title>Parsing JSON</title>
</head>
<body>
<script type="text/javascript">
    var data = '{"name":"Romal","age":20,"dept":"IT"}';

    var jsonObj = JSON.parse(data, function (name, value) {
        return value;
    });
    document.writeln(jsonObj.name);
    for (key in jsonObj) {
        document.writeln("<br/>" + key + " : " + jsonObj[key]);
    }

    // reviver that drops the key "sem"
    var jsonObj = JSON.parse(data, function (name, value) {
        if (name == "sem") {
            return undefined;
        } else {
            return value;
        }
    });
    for (key in jsonObj) {
        document.writeln("<br/>" + key + " : " + jsonObj[key]);
    }
</script>
</body>
</html>
